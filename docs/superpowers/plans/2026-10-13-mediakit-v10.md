```python
"""Cliente do VectorCraft sem tela (vectorcraft-cli mcp --headless), um processo por chamador.
Uso: v = VC(); v.ex("shape.rectangle", {...}); v.close()"""
import json
import subprocess
from pathlib import Path

from mediakit import config


class VCError(Exception):
    pass


def versao_travada():
    return config.VERSAO_VC.read_text().strip()


def versao_instalada(repo):
    return subprocess.run(["git", "-C", str(repo), "rev-parse", "HEAD"], capture_output=True, text=True,
                          check=True).stdout.strip()


def conferir_versao(repo=None):
    repo = Path(repo or config.VC_REPO)
    tem, quer = versao_instalada(repo), versao_travada()
    if tem != quer:
        raise VCError(f"VectorCraft em {repo} está no commit {tem[:10]}, mas o testado é {quer[:10]}. "
                      f"Rode scripts/montar_vectorcraft.sh para voltar à versão testada.")


class VC:
    def __init__(self, read_root="/", write_root="/"):
        self.p = subprocess.Popen([str(config.VC_CLI), "mcp", "--headless", "--automation-read-root", read_root,
                                   "--automation-write-root", write_root],
                                  stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL,
                                  text=True, bufsize=1)
        self.n = 0
        self._rpc("initialize", {"protocolVersion": "2025-06-18", "capabilities": {},
                                 "clientInfo": {"name": "mediakit", "version": "1"}})
        self.p.stdin.write(json.dumps({"jsonrpc": "2.0", "method": "notifications/initialized"}) + "\n")
        self.p.stdin.flush()

    def _rpc(self, method, params):
        self.n += 1
        self.p.stdin.write(json.dumps({"jsonrpc": "2.0", "id": self.n, "method": method, "params": params}) + "\n")
        self.p.stdin.flush()
        while True:
            line = self.p.stdout.readline()
            if not line:
                raise VCError("vectorcraft-cli encerrou")
            m = json.loads(line)
            if m.get("id") == self.n:
                if "error" in m:
                    raise VCError(f"{method}: {m['error']}")
                return m["result"]

    def tool(self, name, args):
        r = self._rpc("tools/call", {"name": name, "arguments": args})
        txt = "".join(c.get("text", "") for c in r.get("content", []) if c.get("type") == "text")
        if r.get("isError"):
            raise VCError(f"{name} {json.dumps(args, ensure_ascii=False)[:200]} -> {txt[:400]}")
        try:
            return json.loads(txt)
        except ValueError:
            return txt

    def ex(self, command, params=None):
        return self.tool("run_command", {"command": command, "params": params or {}})

    def close(self):
        try:
            self.p.stdin.close()
            self.p.wait(timeout=10)
        except Exception:
            self.p.kill()
```

```python
"""Primitivas de página sobre o VectorCraft. Cada Pagina é um artboard; coordenadas em unidades da página (1920 × 1080)."""

PRETO, BRANCO = "#000000", "#FFFFFF"
NEGRITO = ("Helvetica Neue", "Bold")


def _achar(no, oid):
    if no.get("id") == oid and no.get("kind") != "Layer":
        return no
    for c in no.get("children", []) or []:
        r = _achar(c, oid)
        if r:
            return r
    return None


def bounds(ex, oid):
    d = ex("document.inspect", {"depth": 3})
    for camada in d["layers"]:
        r = _achar(camada, oid)
        if r:
            return r["bounds"]
    raise KeyError(oid)


class Pagina:
    def __init__(self, ex, idx, pw=1920, ph=1080, gap=2000):
        self.ex, self.i, self.pw, self.ph = ex, idx, pw, ph
        self.ox = idx * gap
        if idx > 0:
            ex("artboard.new", {"x": self.ox, "y": 0, "width": pw, "height": ph})

    def _pintar(self, oid, fill=None, stroke=None, sw=None, opacity=None):
        ex = self.ex
        ex("select.set", {"ids": [oid]})
        ex("paint.setFill", {"none": True} if fill is None else {"color": fill})
        if stroke is None:
            ex("paint.setStroke", {"none": True})
        else:
            ex("paint.setStroke", {"color": stroke})
            ex("stroke.set", {"weight": sw or 1})
        if opacity is not None:
            ex("transparency.set", {"ids": [oid], "opacity": opacity})
        ex("select.none")
        return oid

    def rect(self, x, y, w, h, fill=PRETO, stroke=None, sw=None, radius=None, opacity=None):
        p = {"x": self.ox + x, "y": y, "width": w, "height": h}
        if radius:
            p["radius"] = radius
        return self._pintar(self.ex("shape.rectangle", p)["id"], fill, stroke, sw, opacity)

    def ellipse(self, cx, cy, rx, ry, fill=PRETO, stroke=None, sw=None):
        oid = self.ex("shape.ellipse", {"x": self.ox + cx - rx, "y": cy - ry, "width": 2 * rx, "height": 2 * ry})["id"]
        return self._pintar(oid, fill, stroke, sw)

    def path(self, d, fill=None, stroke=None, sw=None, opacity=None):
        return self._pintar(self.ex("path.create", {"d": d})["id"], fill, stroke, sw, opacity)

    def poly_d(self, pts, closed=True):
        s = "M" + " L".join(f"{self.ox + x:.2f},{y:.2f}" for x, y in pts)
        return s + (" Z" if closed else "")

    def image(self, file, x, y, w, h):
        oid = self.ex("file.place", {"path": str(file), "link": False})["ids"][0]
        self.ex("select.set", {"ids": [oid]})
        self.ex("object.setBounds", {"x": self.ox + x, "y": y, "width": w, "height": h, "reference": 0})
        self.ex("select.none")
        return oid

    def text(self, x, y, s, font=NEGRITO, size=40, color=BRANCO, tracking=None, leading=None,
             justify=None, width=None, features=None):
        p = {"x": self.ox + x, "y": y, "text": s, "font": font[0], "style": font[1], "size": size, "color": color}
        if width:
            p["area"] = {"width": width, "height": 2000, "fit": "autoHeight"}
        oid = self.ex("text.create", p)["id"]
        st = {"ids": [oid]}
        for k, v in (("tracking", tracking), ("leading", leading), ("justify", justify), ("features", features)):
            if v is not None:
                st[k] = v
        if len(st) > 1:
            self.ex("text.setStyle", st)
        return oid

    def fit(self, x, y, s, max_w, font=NEGRITO, size=300, color=BRANCO, tracking=None, leading=None,
            justify=None, max_h=None):
        """Cria o texto e escala o corpo até a linha mais longa ter max_w (e caber em max_h).
        Atenção: o `size` passado é só o ponto de partida; quem manda no tamanho final é max_w."""
        oid = self.text(x, y, s, font, size, color, tracking, leading, justify)
        b = bounds(self.ex, oid)
        k = max_w / b["width"]
        if max_h:
            k = min(k, max_h / b["height"])
        st = {"ids": [oid], "size": size * k}
        if leading is not None:
            st["leading"] = leading * k
        self.ex("text.setStyle", st)
        return oid, size * k
```

"""Tratamento raster das fotos ANTES de entrarem no vetor, com a régua de resolução da página exportada."""
from pathlib import Path

import numpy as np
from PIL import Image, ImageEnhance, ImageOps

from mediakit import config

AVISOS = []


def ppi_na_pagina(px_largura, unidades_largura):
    polegadas = unidades_largura * config.ESCALA_PDF / 72
    return px_largura / polegadas


def preparar_foto(origem, destino, w, h, modo="bw", foco=(0.5, 0.5), rastro=0, escuro=1.0):
    """Recorta no formato da caixa w × h (unidades da página) pelo foco, nunca amplia, e avisa abaixo de 96 ppi.
    modo: "bw" (preto e branco com contraste automático), "warm" (contraste +8 %) ou "none".
    rastro: largura em px do rastro de movimento (nítido à esquerda, rastro à direita); 0 desliga."""
    im = Image.open(origem).convert("RGB")
    ar = w / h
    iw, ih = im.size
    cw, ch = (ih * ar, ih) if iw / ih > ar else (iw, iw / ar)
    cx = min(max(foco[0] * iw, cw / 2), iw - cw / 2)
    cy = min(max(foco[1] * ih, ch / 2), ih - ch / 2)
    im = im.crop((int(cx - cw / 2), int(cy - ch / 2), int(cx + cw / 2), int(cy + ch / 2)))
    sc = min(1.0, max(w, h) * 1.2 / max(im.size))
    im = im.resize((int(im.size[0] * sc), int(im.size[1] * sc)), Image.LANCZOS)
    if rastro:
        a = np.asarray(im).astype("float32")
        passos = range(-rastro, rastro + 1, 2)
        mov = sum(np.roll(a, k, axis=1) for k in passos) / len(passos)
        mascara = np.linspace(0, 1, a.shape[1])[None, :, None] ** 1.5
        im = Image.fromarray((a * (1 - mascara) + mov * mascara).astype("uint8"))
    if modo == "bw":
        im = ImageOps.autocontrast(ImageOps.grayscale(im), cutoff=1).convert("RGB")
    elif modo == "warm":
        im = ImageEnhance.Contrast(im).enhance(1.08)
    if escuro != 1.0:
        im = ImageEnhance.Brightness(im).enhance(escuro)
    Path(destino).parent.mkdir(parents=True, exist_ok=True)
    im.save(destino, quality=86)
    ppi = ppi_na_pagina(im.size[0], w)
    aviso = None
    if ppi < config.PPI_MINIMO:
        aviso = (f"{Path(origem).name}: {ppi:.0f} ppi numa caixa de {w:.0f} × {h:.0f} (régua {config.PPI_MINIMO}). "
                 f"Use uma caixa menor, outra foto, ou aceite com o Ugo.")
        AVISOS.append(aviso)
    return {"caminho": Path(destino), "ppi": round(ppi, 1), "aviso": aviso}
```

"""Qual cena entra no kit (gerada com o logo, se aprovada no olho; senão a composta) e quanto da cota foi gasto."""
import json
import sys
from datetime import datetime
from pathlib import Path


def _ler(p, padrao):
    return json.loads(p.read_text()) if p.exists() else padrao


def escolher_cena(acervo, composta, gerada):
    acervo = Path(acervo)
    escolha = _ler(acervo / "escolha-cenas.json", {})
    if escolha.get(gerada) == "usar" and (acervo / gerada).exists():
        return acervo / gerada, "gerada"
    return acervo / composta, "composta"


def marcar(acervo, gerada, decisao):
    if decisao not in ("usar", "descartar"):
        raise ValueError("decisão é 'usar' ou 'descartar'")
    p = Path(acervo) / "escolha-cenas.json"
    escolha = _ler(p, {})
    escolha[gerada] = decisao
    p.write_text(json.dumps(escolha, ensure_ascii=False, indent=1) + "\n")


def geracoes(acervo):
    return _ler(Path(acervo) / "cota.json", [])


def registrar_geracao(acervo, id_, quando=None):
    p = Path(acervo) / "cota.json"
    lista = _ler(p, [])
    lista.append({"id": id_, "quando": quando or datetime.now().isoformat(timespec="minutes")})
    p.write_text(json.dumps(lista, ensure_ascii=False, indent=1) + "\n")
    return len(lista)


if __name__ == "__main__":
    if len(sys.argv) == 4 and sys.argv[1] == "registrar":
        print(registrar_geracao(sys.argv[2], sys.argv[3]))
    else:
        sys.exit("uso: python3 -m mediakit.cenas registrar <acervo> <id>")
```

"""Texto encostando em texto, medido pela tinta de cada letra (TEXT_ACCURATE_BBOXES do MuPDF), não pela caixa da fonte."""
import itertools

import fitz


def _linhas(pagina):
    d = pagina.get_text("rawdict", flags=fitz.TEXT_ACCURATE_BBOXES)
    for bloco in d["blocks"]:
        for linha in bloco.get("lines", []):
            letras = [c for s in linha["spans"] for c in s["chars"] if c["c"].strip()]
            if letras:
                yield "".join(c["c"] for c in letras), [fitz.Rect(c["bbox"]) for c in letras]


def sobreposicoes(pdf, area_minima=1.0):
    achados = []
    for n, pagina in enumerate(fitz.open(pdf), 1):
        for (ta, ra), (tb, rb) in itertools.combinations(list(_linhas(pagina)), 2):
            pares = sum(1 for a in ra for b in rb if abs(a & b) > area_minima)
            if pares:
                achados.append({"pagina": n, "a": ta, "b": tb, "pares": pares})
    return achados
```

"""Conteúdo do PDF contra a ficha aprovada: número que não está na ficha e travessão."""
import re

import fitz

_NUM = re.compile(r"\d+(?:[./-]\d+)*%?")
_TRAVESSAO = re.compile("(?s).{0,20}[\u2014\u2013].{0,20}")


def numeros(texto):
    return set(_NUM.findall(texto))


def _paginas(pdf):
    for n, p in enumerate(fitz.open(pdf), 1):
        yield n, p.get_text()


def numeros_fora_da_ficha(pdf, ficha_md):
    aceitos = numeros(open(ficha_md, encoding="utf-8").read())
    fora = []
    for n, texto in _paginas(pdf):
        for num in sorted(numeros(texto) - aceitos):
            fora.append({"pagina": n, "numero": num})
    return fora


def travessoes(pdf):
    return [{"pagina": n, "trecho": m.group(0).strip()} for n, texto in _paginas(pdf) for m in _TRAVESSAO.finditer(texto)]

"""As camadas automáticas da checagem (conteúdo, arquivo, tinta) num relatório só; e o prompt do revisor independente.
Uso: python3 -m mediakit.checagem.checar <pdf> <ficha.md>   (sai 1 se houver falha)"""
import subprocess
import sys
from pathlib import Path

from mediakit import config
from mediakit.checagem import conteudo, tinta


def _validar_arquivo(pdf):
    r = subprocess.run([sys.executable, str(config.VALIDAR_PDF), str(pdf)], capture_output=True, text=True)
    return r.returncode == 0, (r.stdout + r.stderr).strip()


def checar(pdf, ficha, avisos_foto=()):
    falhas, avisos = [], []
    for s in tinta.sobreposicoes(pdf):
        falhas.append(f"p{s['pagina']}: sobreposição de texto: '{s['a'][:40]}' encosta em '{s['b'][:40]}'")
    for t in conteudo.travessoes(pdf):
        falhas.append(f"p{t['pagina']}: travessão em '{t['trecho']}'")
    ok, saida = _validar_arquivo(pdf)
    if not ok:
        falhas.append("validar_pdf reprovou o arquivo (ver saída)")
    for n in conteudo.numeros_fora_da_ficha(pdf, ficha):
        avisos.append(f"p{n['pagina']}: número '{n['numero']}' não está na ficha")
    avisos.extend(avisos_foto)
    return {"falhas": falhas, "avisos": avisos, "arquivo": saida}


def relatorio_md(r):
    linhas = ["## Checagem automática", "", f"Falhas: {len(r['falhas'])} · Avisos: {len(r['avisos'])}", ""]
    linhas += [f"- FALHA {f}" for f in r["falhas"]] + [f"- aviso {a}" for a in r["avisos"]]
    linhas += ["", "### Arquivo (validar_pdf)", "", "```", r["arquivo"], "```"]
    return "\n".join(linhas) + "\n"


def prompt_revisor(paginas_dir, ficha, referencias, logo):
    modelo = (Path(__file__).parent / "revisor.md").read_text(encoding="utf-8")
    return modelo.format(paginas=f"{paginas_dir}/pag-*.png", ficha=ficha,
                         referencias="; ".join(str(r) for r in referencias), logo=logo)


if __name__ == "__main__":
    if len(sys.argv) != 3:
        sys.exit("uso: python3 -m mediakit.checagem.checar <pdf> <ficha.md>")
    res = checar(sys.argv[1], sys.argv[2])
    print(relatorio_md(res))
    sys.exit(1 if res["falhas"] else 0)
```

Ordem pelo ranking visual dos 87 decks do repertório (inventario/tags-a*.json, 13/10/2026)."""
import json
import re
import shutil
from pathlib import Path

from mediakit import config

ORDEM = ["camisa", "celular-post", "cartaz", "kit-sacola", "copo", "backdrop", "notebook", "outdoor", "credencial",
         "embalagem", "portico", "bone", "wind-banner", "tenda", "celular-app", "numero-de-peito", "placa", "telao",
         "medalha", "stories", "estande"]


def _raiz(raiz):
    return Path(raiz) if raiz else config.RAIZ / "biblioteca" / "pecas"


def _slug(s):
    return re.sub(r"[^a-z0-9]+", "-", s.lower()).strip("-")


def listar(peca=None, raiz=None):
    indice = _raiz(raiz) / "indice.json"
    itens = json.loads(indice.read_text()) if indice.exists() else []
    return [i for i in itens if peca is None or i["peca"] == peca]


def registrar(peca, arquivo, marca, kit, ambiente="", raiz=None):
    if peca not in ORDEM:
        raise ValueError(f"peça '{peca}' não está na lista: {', '.join(ORDEM)}")
    base = _raiz(raiz)
    destino = base / peca / f"{_slug(marca)}-{_slug(kit)}{Path(arquivo).suffix}"
    destino.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(arquivo, destino)
    itens = listar(raiz=base)
    itens = [i for i in itens if i["arquivo"] != str(destino.relative_to(base))]
    itens.append({"peca": peca, "arquivo": str(destino.relative_to(base)), "marca": marca, "kit": kit, "ambiente": ambiente})
    (base / "indice.json").write_text(json.dumps(itens, ensure_ascii=False, indent=1) + "\n")
    return destino

"""Biblioteca de referências: lê o repertório do cofre e acrescenta referência nova (nunca edita a existente)."""
import json
import re
import shutil
import sys
from datetime import date
from pathlib import Path

import fitz
from PIL import Image

from mediakit import config

_ID = re.compile(r"^[a-z0-9][a-z0-9-]*$")
_IMAGENS = (".jpg", ".jpeg", ".png", ".webp")


def _raiz(r):
    return Path(r) if r else config.COFRE_MEDIAKIT / "repertorio"


def _registro(r):
    return Path(r) if r else config.RAIZ / "biblioteca" / "referencias.json"


def _novas(registro):
    p = _registro(registro)
    return json.loads(p.read_text()) if p.exists() else []


def listar(raiz_repertorio=None, registro=None):
    novas = {n["id"] for n in _novas(registro)}
    cap = _raiz(raiz_repertorio) / "capturas"
    itens = []
    for pasta in sorted(p for p in cap.iterdir() if p.is_dir()) if cap.exists() else []:
        n = sum(1 for f in pasta.iterdir() if f.suffix.lower() in _IMAGENS)
        itens.append({"id": pasta.name, "pasta": pasta, "imagens": n, "origem": "nova" if pasta.name in novas else "repertorio"})
    return itens


def existe(id_, raiz_repertorio=None):
    return (_raiz(raiz_repertorio) / "capturas" / id_).exists()


def adicionar(id_, arquivos, origem_texto, raiz_repertorio=None, registro=None):
    if not _ID.match(id_):
        raise ValueError(f"id inválido: {id_!r} (use a-z, 0-9 e hífen)")
    if existe(id_, raiz_repertorio):
        raise ValueError(f"referência {id_} já existe; escolha outro id")
    pasta = _raiz(raiz_repertorio) / "capturas" / id_
    pasta.mkdir(parents=True)
    n = 0
    for a in map(Path, arquivos):
        if a.suffix.lower() == ".pdf":
            for pg in fitz.open(a):
                n += 1
                pg.get_pixmap(dpi=110).save(pasta / f"{n:02d}.jpg")
        else:
            n += 1
            Image.open(a).convert("RGB").save(pasta / f"{n:02d}.jpg", quality=88)
    reg = _registro(registro)
    reg.parent.mkdir(parents=True, exist_ok=True)
    lista = _novas(registro) + [{"id": id_, "origem_texto": origem_texto, "quando": date.today().isoformat(), "imagens": n}]
    reg.write_text(json.dumps(lista, ensure_ascii=False, indent=1) + "\n")
    return pasta


if __name__ == "__main__":
    if len(sys.argv) >= 4 and sys.argv[1] == "adicionar":
        print(adicionar(sys.argv[2], sys.argv[4:], sys.argv[3]))
    elif sys.argv[1:] == ["listar"]:
        for i in listar():

Uso: python3 scripts/rodar_receitas.py [--gravar-base]"""
import json
import subprocess
import sys
from pathlib import Path

RAIZ = Path(__file__).resolve().parents[1]
BASE = RAIZ / "receitas" / "_linha-de-base.json"


def rodar():
    res = {}
    for t in sorted((RAIZ / "receitas").glob("R*/teste.py")):
        try:
            r = subprocess.run([sys.executable, str(t)], cwd=t.parent, capture_output=True, text=True, timeout=120)
            res[t.parent.name] = r.returncode == 0
        except subprocess.TimeoutExpired:
            res[t.parent.name] = False
        print(t.parent.name, "ok" if res[t.parent.name] else "FALHOU", flush=True)
    return res


if __name__ == "__main__":
    res = rodar()
    if "--gravar-base" in sys.argv:
        BASE.write_text(json.dumps(res, indent=1, sort_keys=True) + "\n")
        print(f"linha de base: {sum(res.values())} de {len(res)} rodam")
        sys.exit(0)
    base = json.loads(BASE.read_text())
